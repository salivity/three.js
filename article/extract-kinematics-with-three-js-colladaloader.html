Extract Kinematics with Three.js ColladaLoader

Extracting and evaluating complex robotic kinematics chains from legacy COLLADA (.dae) files requires parsing the underlying kinematic definitions, mapping abstract joints to 3D scene objects, and computing coordinate transforms. This guide covers how to ingest legacy digital asset exchange files using Three.js's ColladaLoader, access the parsed kinematics dictionary, map revolute and prismatic joints to their corresponding scene graph transforms, and evaluate forward kinematics across the chain.

1. Ingesting Legacy COLLADA Assets

Legacy COLLADA files featuring robotic setups typically store kinematics under <kinematics_model> and <articulated_system> extensions. When ColladaLoader processes these files, it creates both a standard Three.js visual hierarchy (collada.scene) and a specialized kinematics property (collada.kinematics).

import { ColladaLoader } from 'three/addons/loaders/ColladaLoader.js';

const loader = new ColladaLoader();

loader.load('path/to/legacy_robot.dae', (collada) => {
    const robotScene = collada.scene;
    const kinematics = collada.kinematics;

    scene.add(robotScene);

    if (kinematics) {
        setupKinematicsChain(robotScene, kinematics);
    } else {
        console.warn('No kinematics model detected in Collada payload.');
    }
});

2. Inspecting the Kinematics Schema

The collada.kinematics object holds the definitions required to animate the robot:

  • joints: An array or dictionary of joints, detailing types (e.g., revolute, prismatic), axis vectors, and limits (min, max).
  • links: Structural connections connecting one joint to the next.
  • setJointValue(jointIndexOrName, value): Built-in helper generated by ColladaLoader to dynamically set joint transforms.

To manually control or evaluate the hierarchy, iterate over the parsed joints:

function inspectKinematics(kinematics) {
    kinematics.joints.forEach((joint, index) => {
        console.log(`Joint [${index}]: ${joint.name}`);
        console.log(`Type: ${joint.type}`);
        console.log(`Axis:`, joint.axis);
        console.log(`Limits: [${joint.limits.min}, ${joint.limits.max}]`);
    });
}

3. Mapping Abstract Joints to Scene Nodes

Legacy files often separate visual nodes from structural kinematic joints, binding them via identifier references (node target names or URIs). If you choose not to use the loader’s internal setJointValue helper, establish a direct mapping between the kinematic joints and THREE.Object3D instances:

function mapJointsToNodes(robotScene, kinematics) {
    const jointNodeMap = new Map();

    kinematics.joints.forEach((joint) => {
        // Collada targets can be node names, IDs, or scoped SID paths
        const targetNode = robotScene.getObjectByName(joint.target || joint.name);

        if (targetNode) {
            jointNodeMap.set(joint.name, {
                definition: joint,
                node: targetNode,
                zeroMatrix: targetNode.matrix.clone() // Cache base reference pose
            });
        }
    });

    return jointNodeMap;
}

4. Evaluating Forward Kinematics

To resolve the forward kinematics manually:

  1. Calculate Joint Displacement: Apply rotation around the joint axis for revolute joints, or translation along the joint axis for prismatic joints.
  2. Apply Local Transform: Update the node's local transform using its cached origin transformation.
  3. Propagate Matrices: Enforce recursive matrix recalculations down the scene graph to compute the world position of every link, including the end-effector.
import * as THREE from 'three';

function applyJointPose(mappedJoint, value) {
    const { definition, node, zeroMatrix } = mappedJoint;
    const { type, axis, limits } = definition;

    // Clamp value to defined limits
    const clampedValue = Math.max(limits.min, Math.min(limits.max, value));
    
    // Reset to base local transform
    node.matrix.copy(zeroMatrix);

    const deltaTransform = new THREE.Matrix4();

    if (type === 'revolute') {
        const axisVector = new THREE.Vector3(axis.x, axis.y, axis.z).normalize();
        deltaTransform.makeRotationAxis(axisVector, THREE.MathUtils.degToRad(clampedValue));
    } else if (type === 'prismatic') {
        const axisVector = new THREE.Vector3(axis.x, axis.y, axis.z).normalize();
        deltaTransform.makeTranslation(
            axisVector.x * clampedValue,
            axisVector.y * clampedValue,
            axisVector.z * clampedValue
        );
    }

    // Multiply local transform
    node.matrix.multiply(deltaTransform);
    node.matrix.decompose(node.position, node.quaternion, node.scale);
}

5. Computing the End-Effector Transform

Once all target joint values are applied, update the world matrix across the root robotic assembly to extract the final end-effector position and orientation:

function evaluateEndEffector(robotScene, endEffectorNodeName) {
    // Force full hierarchical matrix update
    robotScene.updateMatrixWorld(true);

    const endEffectorNode = robotScene.getObjectByName(endEffectorNodeName);
    if (!endEffectorNode) return null;

    const position = new THREE.Vector3();
    const quaternion = new THREE.Quaternion();
    const scale = new THREE.Vector3();

    endEffectorNode.matrixWorld.decompose(position, quaternion, scale);

    return {
        position,
        quaternion,
        transformMatrix: endEffectorNode.matrixWorld
    };
}
