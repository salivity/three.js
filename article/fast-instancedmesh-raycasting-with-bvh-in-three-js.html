Fast InstancedMesh Raycasting with BVH in Three.js

Raycasting against a THREE.InstancedMesh containing thousands of elements causes severe performance bottlenecks because Three.js tests every instance linearly (\(O(N)\)). This guide explains how to construct a custom, instance-level Bounding Volume Hierarchy (BVH) that partitions instance bounding boxes into a binary tree. By replacing linear iteration with hierarchical bounding box intersection tests, raycasting complexity drops to \(O(\log N)\), enabling real-time interactions with dense scenes.

1. The Core Architecture

Standard mesh BVHs (like three-mesh-bvh) partition the triangles of a single geometry. For an InstancedMesh, geometry is duplicated across many transformation matrices. An instance-level BVH partitions the transformed bounding volumes of these instances rather than individual triangles.

The BVH consists of:

  • Nodes: Each node holds a precomputed Axis-Aligned Bounding Box (THREE.Box3).
  • Internal Nodes: Contain references to left and right child nodes.
  • Leaf Nodes: Contain a list of instance indices that fall within that node's spatial boundary.

2. Extracting Instance Bounding Boxes

Before building the tree, compute the local axis-aligned bounding box for each instance by transforming the base geometry's bounding box with the instance matrix.

import * as THREE from 'three';

function computeInstanceBoxes(instancedMesh) {
  const geometry = instancedMesh.geometry;
  if (!geometry.boundingBox) geometry.computeBoundingBox();

  const baseBox = geometry.boundingBox;
  const count = instancedMesh.count;
  const instanceBoxes = [];
  const matrix = new THREE.Matrix4();
  const transformedBox = new THREE.Box3();

  for (let i = 0; i < count; i++) {
    instancedMesh.getMatrixAt(i, matrix);
    transformedBox.copy(baseBox).applyMatrix4(matrix);
    
    // Store index alongside box and center for partitioning
    const center = new THREE.Vector3();
    transformedBox.getCenter(center);

    instanceBoxes.push({
      index: i,
      box: transformedBox.clone(),
      center: center
    });
  }

  return instanceBoxes;
}

3. Recursive BVH Tree Construction

A top-down recursive spatial split partitions instances along the longest axis of their collective bounding box.

class BVHNode {
  constructor() {
    this.box = new THREE.Box3();
    this.left = null;
    this.right = null;
    this.instanceIndices = null; // Populated only on leaf nodes
  }
}

function buildBVH(items, maxLeafSize = 8) {
  const node = new BVHNode();

  // Calculate the encapsulating box for all items in this node
  for (let i = 0; i < items.length; i++) {
    node.box.union(items[i].box);
  }

  // Base case: create a leaf node if items are within threshold
  if (items.length <= maxLeafSize) {
    node.instanceIndices = items.map(item => item.index);
    return node;
  }

  // Find the longest axis to split along
  const size = new THREE.Vector3();
  node.box.getSize(size);
  let axis = 'x';
  if (size.y > size.x && size.y > size.z) axis = 'y';
  else if (size.z > size.x && size.z > size.y) axis = 'z';

  // Sort items by centroid on the chosen axis
  items.sort((a, b) => a.center[axis] - b.center[axis]);

  // Split items evenly (median split)
  const mid = Math.floor(items.length / 2);
  const leftItems = items.slice(0, mid);
  const rightItems = items.slice(mid);

  node.left = buildBVH(leftItems, maxLeafSize);
  node.right = buildBVH(rightItems, maxLeafSize);

  return node;
}

4. BVH Ray Traversal

When raycasting, test the ray against the node's Box3. If the ray does not intersect, prune the entire branch. If it is a leaf, test the ray against the individual instances inside.

const _ray = new THREE.Ray();
const _instanceMatrix = new THREE.Matrix4();
const _inverseMatrix = new THREE.Matrix4();
const _intersectionPoint = new THREE.Vector3();

function raycastBVH(node, raycaster, instancedMesh, intersects) {
  // Test ray intersection with the current node's bounding box
  if (!raycaster.ray.intersectsBox(node.box)) {
    return;
  }

  // If internal node, traverse children
  if (!node.instanceIndices) {
    if (node.left) raycastBVH(node.left, raycaster, instancedMesh, intersects);
    if (node.right) raycastBVH(node.right, raycaster, instancedMesh, intersects);
    return;
  }

  // If leaf node, test against candidate instances
  const geometry = instancedMesh.geometry;
  const material = instancedMesh.material;

  for (let i = 0; i < node.instanceIndices.length; i++) {
    const instanceId = node.instanceIndices[i];
    instancedMesh.getMatrixAt(instanceId, _instanceMatrix);

    // Transform ray into instance space
    _inverseMatrix.copy(_instanceMatrix).invert();
    _ray.copy(raycaster.ray).applyMatrix4(_inverseMatrix);

    // Bounding sphere/box pre-check in local space
    if (geometry.boundingSphere && !_ray.intersectsSphere(geometry.boundingSphere)) {
      continue;
    }

    // Precise triangle/geometry intersection in local space
    const localIntersects = [];
    instancedMesh.raycastInstance(_ray, instanceId, localIntersects);

    // Convert local intersections to world space
    for (let j = 0; j < localIntersects.length; j++) {
      const hit = localIntersects[j];
      hit.point.applyMatrix4(_instanceMatrix);
      hit.distance = raycaster.ray.origin.distanceTo(hit.point);
      hit.instanceId = instanceId;
      hit.object = instancedMesh;
      intersects.push(hit);
    }
  }
}

5. Overriding InstancedMesh.raycast

Integrate the BVH directly into the mesh instance to make it compatible with Three.js's standard Raycaster.

// Build the hierarchy
const items = computeInstanceBoxes(instancedMesh);
const bvhRoot = buildBVH(items, 8);

// Attach and override default raycast
instancedMesh.bvh = bvhRoot;
instancedMesh.raycast = function (raycaster, intersects) {
  if (!this.geometry || !this.material) return;
  
  const initialCount = intersects.length;
  raycastBVH(this.bvh, raycaster, this, intersects);

  // Maintain standard Three.js distance sorting for newly added hits
  if (intersects.length > initialCount) {
    intersects.sort((a, b) => a.distance - b.distance);
  }
};

6. Dynamic Updates

If instances move or are updated dynamically:

  • Low-frequency changes: Re-run computeInstanceBoxes and buildBVH. Because instance counts are typically low relative to triangle counts, rebuilding takes only a few milliseconds for tens of thousands of instances.
  • High-frequency updates: Update the bounding boxes up the hierarchy from leaf to root without restructuring the tree branches (refitting), or combine instances into spatial chunks that map to static BVH leaves.
