Control Three.js Camera with Mouse Wheel Events

This article explains how to bind a DOM wheel event listener to control a Three.js camera's zoom or position using scroll wheel delta values. By intercepting mouse wheel input and mapping event.deltaY to camera parameters—such as the Field of View (FOV) or position coordinates—you can implement custom zoom mechanics, handle constraints, and ensure smooth rendering without relying on external control libraries.

1. Attaching the Wheel Event Listener

To manipulate the camera, attach an event listener to the canvas element (renderer.domElement) rather than the global window. This confines the behavior to the Three.js viewport. Pass { passive: false } to allow calling event.preventDefault(), which stops the default browser scroll behavior.

const canvas = renderer.domElement;

canvas.addEventListener('wheel', onDocumentWheel, { passive: false });

function onDocumentWheel(event) {
  event.preventDefault();
  // Handle wheel delta here
}

2. Method A: Zoom via Field of View (FOV)

Modifying the fov property of a PerspectiveCamera simulates zooming like a camera lens. After altering camera.fov, you must call camera.updateProjectionMatrix() to recalculate the projection.

const zoomSpeed = 0.05;
const minFov = 15;
const maxFov = 90;

function onDocumentWheel(event) {
  event.preventDefault();

  // Adjust FOV based on scroll direction
  camera.fov += event.deltaY * zoomSpeed;

  // Clamp the FOV to prevent flipping or extreme distortion
  camera.fov = Math.max(minFov, Math.min(maxFov, camera.fov));

  // Update projection matrix to apply changes
  camera.updateProjectionMatrix();
}

3. Method B: Dolly via Camera Position

Dollies move the camera physically closer to or further from the target along its local z-axis, maintaining the original lens perspective.

const dollySpeed = 0.01;
const minDistance = 2;
const maxDistance = 50;

function onDocumentWheel(event) {
  event.preventDefault();

  // Calculate movement vector along the camera's view direction
  const forward = new THREE.Vector3();
  camera.getWorldDirection(forward);

  // Determine movement step
  const moveStep = event.deltaY * dollySpeed;

  // Calculate potential new position
  const newPosition = camera.position.clone().addScaledVector(forward, -moveStep);

  // Distance check from scene center (0, 0, 0)
  const distance = newPosition.length();
  if (distance >= minDistance && distance <= maxDistance) {
    camera.position.copy(newPosition);
  }
}

4. Complete Implementation Example

The following script initializes a scene, renderer, and camera, then connects the wheel event to modify distance along the Z-axis:

import * as THREE from 'three';

// Scene and Camera Setup
const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
camera.position.z = 10;

// Renderer Setup
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

// Add an object to view
const geometry = new THREE.BoxGeometry();
const material = new THREE.MeshBasicMaterial({ color: 0x00ff00, wireframe: true });
const cube = new THREE.Mesh(geometry, material);
scene.add(cube);

// Custom Wheel Listener
const zoomSpeed = 0.02;
const minZ = 2;
const maxZ = 30;

renderer.domElement.addEventListener('wheel', (event) => {
  event.preventDefault();

  // Adjust Z position based on deltaY
  camera.position.z += event.deltaY * zoomSpeed;

  // Clamp within bounds
  camera.position.z = Math.min(Math.max(camera.position.z, minZ), maxZ);
}, { passive: false });

// Animation Loop
function animate() {
  requestAnimationFrame(animate);
  cube.rotation.x += 0.01;
  cube.rotation.y += 0.01;
  renderer.render(scene, camera);
}

animate();

5. Cleaning Up Listeners

If your Three.js instance unmounts (such as in single-page applications built with React, Vue, or Angular), remove the listener to avoid memory leaks:

function cleanup() {
  renderer.domElement.removeEventListener('wheel', onDocumentWheel);
  renderer.dispose();
}
