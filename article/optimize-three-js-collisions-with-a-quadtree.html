Optimize Three.js Collisions with a Quadtree

Brute-force collision detection in Three.js scales quadratically (\(O(n^2)\)), causing frame rates to collapse when handling thousands of moving meshes. Implementing a quadtree drastically improves rendering performance by recursively partitioning spatial coordinates into four quadrants, narrowing candidate checks down to an average of \(O(n \log n)\) complexity. This article explains how to build a lightweight 2D spatial quadtree mapped to 3D coordinates, populate it with Three.js object data, and run optimized collision queries inside your render loop.

Understanding the Quadtree Structure

In a standard 3D scene where entities move primarily across a ground plane (X and Z axes), a 2D quadtree is ideal. Instead of checking every object against every other object, the space is divided into bounding boxes. A node subdivides into four sub-nodes (Northwest, Northeast, Southwest, Southeast) only when the number of objects it contains exceeds a defined capacity threshold.

Step 1: Create Bounding Boxes and the Quadtree Class

First, define a 2D bounding area using an axis-aligned bounding box (AABB) centered on the X and Z axes, followed by the core Quadtree class.

class BoundingBox2D {
  constructor(x, z, halfWidth, halfDepth) {
    this.x = x;
    this.z = z;
    this.halfWidth = halfWidth;
    this.halfDepth = halfDepth;
  }

  contains(point) {
    return (
      point.x >= this.x - this.halfWidth &&
      point.x <= this.x + this.halfWidth &&
      point.z >= this.z - this.halfDepth &&
      point.z <= this.z + this.halfDepth
    );
  }

  intersects(range) {
    return !(
      range.x - range.halfWidth > this.x + this.halfWidth ||
      range.x + range.halfWidth < this.x - this.halfWidth ||
      range.z - range.halfDepth > this.z + this.halfDepth ||
      range.z + range.halfDepth < this.z - this.halfDepth
    );
  }
}

class Quadtree {
  constructor(boundary, capacity = 4) {
    this.boundary = boundary;
    this.capacity = capacity;
    this.objects = [];
    this.divided = false;
  }

  subdivide() {
    const { x, z, halfWidth, halfDepth } = this.boundary;
    const nextW = halfWidth / 2;
    const nextD = halfDepth / 2;

    this.northwest = new Quadtree(new BoundingBox2D(x - nextW, z - nextD, nextW, nextD), this.capacity);
    this.northeast = new Quadtree(new BoundingBox2D(x + nextW, z - nextD, nextW, nextD), this.capacity);
    this.southwest = new Quadtree(new BoundingBox2D(x - nextW, z + nextD, nextW, nextD), this.capacity);
    this.southeast = new Quadtree(new BoundingBox2D(x + nextW, z + nextD, nextW, nextD), this.capacity);

    this.divided = true;
  }

  insert(item) {
    if (!this.boundary.contains(item.position)) {
      return false;
    }

    if (this.objects.length < this.capacity && !this.divided) {
      this.objects.push(item);
      return true;
    }

    if (!this.divided) {
      this.subdivide();
    }

    return (
      this.northwest.insert(item) ||
      this.northeast.insert(item) ||
      this.southwest.insert(item) ||
      this.southeast.insert(item)
    );
  }

  query(range, found = []) {
    if (!this.boundary.intersects(range)) {
      return found;
    }

    for (const item of this.objects) {
      if (range.contains(item.position)) {
        found.push(item);
      }
    }

    if (this.divided) {
      this.northwest.query(range, found);
      this.northeast.query(range, found);
      this.southwest.query(range, found);
      this.southeast.query(range, found);
    }

    return found;
  }

  clear() {
    this.objects = [];
    this.divided = false;
    this.northwest = null;
    this.northeast = null;
    this.southwest = null;
    this.southeast = null;
  }
}

Step 2: Integrate the Quadtree into the Three.js Animation Loop

To handle dynamic bodies moving through the scene, the tree must be cleared and rebuilt every frame. Store a collection of Three.js meshes and query nearby entities only within a specific search range around each entity.

import * as THREE from 'three';

// Setup basic Three.js scene
const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

// World dimensions
const WORLD_SIZE = 200;
const OBJECT_COUNT = 3000;
const COLLISION_RADIUS = 1.0;

// Initialize Quadtree covering the world plane
const boundary = new BoundingBox2D(0, 0, WORLD_SIZE / 2, WORLD_SIZE / 2);
const quadtree = new Quadtree(boundary, 8);

// Create entities
const geometry = new THREE.SphereGeometry(COLLISION_RADIUS, 8, 8);
const material = new THREE.MeshBasicMaterial({ color: 0x00ff00 });
const entities = [];

for (let i = 0; i < OBJECT_COUNT; i++) {
  const mesh = new THREE.Mesh(geometry, material.clone());
  mesh.position.set(
    (Math.random() - 0.5) * WORLD_SIZE,
    0,
    (Math.random() - 0.5) * WORLD_SIZE
  );
  mesh.userData.velocity = new THREE.Vector3(
    (Math.random() - 0.5) * 0.2,
    0,
    (Math.random() - 0.5) * 0.2
  );
  scene.add(mesh);
  entities.push(mesh);
}

camera.position.set(0, 150, 150);
camera.lookAt(0, 0, 0);

// Pre-allocate query range to avoid per-frame garbage collection
const queryRange = new BoundingBox2D(0, 0, COLLISION_RADIUS * 2, COLLISION_RADIUS * 2);

function animate() {
  requestAnimationFrame(animate);

  // 1. Reset Quadtree
  quadtree.clear();

  // 2. Update mesh positions and insert into quadtree
  for (let i = 0; i < entities.length; i++) {
    const entity = entities[i];
    entity.position.add(entity.userData.velocity);

    // Boundary bounce
    if (Math.abs(entity.position.x) > WORLD_SIZE / 2) entity.userData.velocity.x *= -1;
    if (Math.abs(entity.position.z) > WORLD_SIZE / 2) entity.userData.velocity.z *= -1;

    // Reset visual state
    entity.material.color.setHex(0x00ff00);

    // Insert payload reference
    quadtree.insert({
      position: { x: entity.position.x, z: entity.position.z },
      mesh: entity
    });
  }

  // 3. Narrow-phase collision query
  for (let i = 0; i < entities.length; i++) {
    const current = entities[i];

    queryRange.x = current.position.x;
    queryRange.z = current.position.z;

    const nearby = quadtree.query(queryRange);

    for (let j = 0; j < nearby.length; j++) {
      const candidate = nearby[j].mesh;

      if (current === candidate) continue;

      const dx = current.position.x - candidate.position.x;
      const dz = current.position.z - candidate.position.z;
      const distanceSq = dx * dx + dz * dz;

      // Check circular distance collision
      if (distanceSq < (COLLISION_RADIUS * 2) ** 2) {
        current.material.color.setHex(0xff0000);
        candidate.material.color.setHex(0xff0000);
      }
    }
  }

  renderer.render(scene, camera);
}

animate();

Performance Considerations

  1. Avoid Garbage Collection: In dynamic loops, avoid allocating new BoundingBox2D instances every frame. Re-use existing search bounds and clear pre-allocated query arrays to prevent GC stutter.
  2. Tune Capacity: Setting node capacity between 8 and 16 generally yields the lowest overhead for real-time rendering. Lower numbers create excessive tree depth; higher numbers result in linear search degradation inside dense nodes.
  3. True 3D Spatial Partitioning: If objects move significantly across the Y-axis as well as X and Z, upgrade the implementation from a Quadtree to an Octree by splitting space into eight volumetric cubes using three dimensions.
