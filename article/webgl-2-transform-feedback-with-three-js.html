WebGL 2 Transform Feedback with Three.js

This guide explains how to obtain the underlying WebGL2RenderingContext from a Three.js WebGLRenderer and leverage it to execute WebGL 2.0 Transform Feedback operations. You will learn how to bypass Three.js abstraction layers safely, configure raw transform feedback shaders and buffers, synchronize state between custom WebGL operations and Three.js, and build a high-performance GPU particle or simulation pass directly inside your render loop.

1. Extracting the WebGL 2.0 Context

To execute WebGL 2.0 operations like Transform Feedback, you must ensure that Three.js initializes a WebGL 2 context, which is the default in modern versions of Three.js. You can access the raw context using the getContext() method:

import * as THREE from 'three';

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

// Extract raw WebGL2 context
const gl = renderer.getContext();

if (!(gl instanceof WebGL2RenderingContext)) {
  throw new Error('WebGL 2.0 is not supported or not enabled.');
}

2. Setting Up Raw Transform Feedback Shaders

Transform feedback requires capturing output varyings from the vertex shader before rasterization. Because Three.js abstracts shader compilation, compiling this specific simulation shader directly with the raw WebGL context provides direct control over gl.transformFeedbackVaryings.

function createTransformFeedbackProgram(gl, vsSource, fsSource, varyings) {
  const vs = gl.createShader(gl.VERTEX_SHADER);
  gl.shaderSource(vs, vsSource);
  gl.compileShader(vs);

  const fs = gl.createShader(gl.FRAGMENT_SHADER);
  gl.shaderSource(fs, fsSource);
  gl.compileShader(fs);

  const program = gl.createProgram();
  gl.attachShader(program, vs);
  gl.attachShader(program, fs);

  // Specify the varyings to record into output buffers
  gl.transformFeedbackVaryings(program, varyings, gl.SEPARATE_ATTRIBS);

  gl.linkProgram(program);
  return program;
}

const simVsSource = `#version 300 es
in vec3 a_position;
in vec3 a_velocity;

out vec3 v_position;
out vec3 v_velocity;

uniform float u_deltaTime;

void main() {
    v_velocity = a_velocity + vec3(0.0, -9.8, 0.0) * u_deltaTime;
    v_position = a_position + v_velocity * u_deltaTime;
}
`;

const simFsSource = `#version 300 es
precision mediump float;
void main() {
    discard; // Fragment processing is not needed for the compute pass
}
`;

const simProgram = createTransformFeedbackProgram(
  gl,
  simVsSource,
  simFsSource,
  ['v_position', 'v_velocity']
);

3. Allocating Ping-Pong Buffers and Feedback Objects

Transform feedback typically utilizes a ping-pong technique where one buffer provides read data while the other captures write data, alternating roles each frame.

const count = 10000;
const positions = new Float32Array(count * 3);
const velocities = new Float32Array(count * 3);

for (let i = 0; i < count * 3; i++) {
  positions[i] = (Math.random() - 0.5) * 10;
  velocities[i] = (Math.random() - 0.5) * 2;
}

// Create buffers
function makeBuffer(gl, data) {
  const buf = gl.createBuffer();
  gl.bindBuffer(gl.ARRAY_BUFFER, buf);
  gl.bufferData(gl.ARRAY_BUFFER, data, gl.DYNAMIC_COPY);
  return buf;
}

const posBufA = makeBuffer(gl, positions);
const posBufB = makeBuffer(gl, positions);
const velBufA = makeBuffer(gl, velocities);
const velBufB = makeBuffer(gl, velocities);

let currentRead = { pos: posBufA, vel: velBufA };
let currentWrite = { pos: posBufB, vel: velBufB };

// Create Transform Feedback object
const transformFeedback = gl.createTransformFeedback();

4. Interfacing Buffers with Three.js

To render the computed data using Three.js materials, wrap one or both of the raw WebGL buffers in a Three.js BufferGeometry.

const geometry = new THREE.BufferGeometry();
const dummyPositions = new Float32Array(count * 3);

// Assign a BufferAttribute to give Three.js the layout
geometry.setAttribute('position', new THREE.BufferAttribute(dummyPositions, 3));

const material = new THREE.PointsMaterial({ size: 2, color: 0x00ffcc });
const points = new THREE.Points(geometry, material);

const scene = new THREE.Scene();
scene.add(points);

const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
camera.position.z = 15;

5. Managing State in the Render Loop

Three.js tracks and caches WebGL state internally. Modifying the state directly with raw WebGL calls will desynchronize the Three.js state tracker. Call renderer.resetState() immediately after your raw WebGL passes to ensure Three.js rebinds its pipeline appropriately.

const posLoc = gl.getAttribLocation(simProgram, 'a_position');
const velLoc = gl.getAttribLocation(simProgram, 'a_velocity');
const dtLoc = gl.getUniformLocation(simProgram, 'u_deltaTime');

let lastTime = performance.now();

function animate() {
  requestAnimationFrame(animate);

  const now = performance.now();
  const delta = (now - lastTime) / 1000;
  lastTime = now;

  // --- Step 1: Raw WebGL Transform Feedback Pass ---
  gl.useProgram(simProgram);
  gl.uniform1f(dtLoc, delta);

  // Bind input attributes
  gl.bindBuffer(gl.ARRAY_BUFFER, currentRead.pos);
  gl.enableVertexAttribArray(posLoc);
  gl.vertexAttribPointer(posLoc, 3, gl.FLOAT, false, 0, 0);

  gl.bindBuffer(gl.ARRAY_BUFFER, currentRead.vel);
  gl.enableVertexAttribArray(velLoc);
  gl.vertexAttribPointer(velLoc, 3, gl.FLOAT, false, 0, 0);

  // Bind output buffers to transform feedback targets
  gl.bindTransformFeedback(gl.TRANSFORM_FEEDBACK, transformFeedback);
  gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, 0, currentWrite.pos);
  gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, 1, currentWrite.vel);

  // Execute simulation pass
  gl.enable(gl.RASTERIZER_DISCARD);
  gl.beginTransformFeedback(gl.POINTS);
  gl.drawArrays(gl.POINTS, 0, count);
  gl.endTransformFeedback();
  gl.disable(gl.RASTERIZER_DISCARD);

  // Unbind feedback targets
  gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, 0, null);
  gl.bindBufferBase(gl.TRANSFORM_FEEDBACK_BUFFER, 1, null);
  gl.bindTransformFeedback(gl.TRANSFORM_FEEDBACK, null);

  // --- Step 2: Swap Buffers ---
  const temp = currentRead;
  currentRead = currentWrite;
  currentWrite = temp;

  // --- Step 3: Synchronize Buffer to Three.js Geometry ---
  const posAttr = geometry.getAttribute('position');
  // Inject the raw buffer into Three.js internal WebGLAttributes cache
  const attrData = renderer.properties.get(posAttr);
  if (attrData && attrData.__webglBuffer) {
    attrData.__webglBuffer = currentRead.pos;
  }

  // --- Step 4: Reset State and Render Scene ---
  renderer.resetState();
  renderer.render(scene, camera);
}
