Motion Blur with Custom ShaderPass in Three.js

Implementing camera motion blur in Three.js involves calculating the per-pixel velocity caused by camera movement and blurring the final render along that velocity vector. This technique uses a custom post-processing ShaderPass within an EffectComposer, reconstructs world positions from a depth buffer, and compares current camera projection matrices with stored matrices from the previous frame. This guide covers setting up the depth render target, constructing the velocity-based motion blur shader, and updating the camera matrices frame by frame.

1. Configure the Depth Render Target

To reconstruct world positions in a fragment shader, the scene's depth must be captured. Configure a WebGLRenderTarget equipped with a DepthTexture:

import * as THREE from 'three';
import { EffectComposer } from 'three/examples/jsm/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/examples/jsm/postprocessing/RenderPass.js';
import { ShaderPass } from 'three/examples/jsm/postprocessing/ShaderPass.js';

const width = window.innerWidth;
const height = window.innerHeight;

const renderTarget = new THREE.WebGLRenderTarget(width, height, {
  minFilter: THREE.LinearFilter,
  magFilter: THREE.LinearFilter,
  format: THREE.RGBAFormat,
  type: THREE.HalfFloatType,
  depthTexture: new THREE.DepthTexture(width, height)
});
renderTarget.depthTexture.format = THREE.DepthFormat;
renderTarget.depthTexture.type = THREE.UnsignedIntType;

const composer = new EffectComposer(renderer, renderTarget);
const renderPass = new RenderPass(scene, camera);
composer.addPass(renderPass);

2. Define the Motion Blur Shader

The custom shader reconstructs the current pixel's world position from screen UV and depth. It then projects that position into the previous frame's clip space to determine the pixel's motion vector, finally taking multiple samples along that vector.

const MotionBlurShader = {
  uniforms: {
    tDiffuse: { value: null },
    tDepth: { value: null },
    uInverseViewProjectionMatrix: { value: new THREE.Matrix4() },
    uPreviousViewProjectionMatrix: { value: new THREE.Matrix4() },
    uVelocityScale: { value: 1.0 },
    uSamples: { value: 16 }
  },
  vertexShader: `
    varying vec2 vUv;
    void main() {
      vUv = uv;
      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    }
  `,
  fragmentShader: `
    uniform sampler2D tDiffuse;
    uniform sampler2D tDepth;
    uniform mat4 uInverseViewProjectionMatrix;
    uniform mat4 uPreviousViewProjectionMatrix;
    uniform float uVelocityScale;
    uniform int uSamples;

    varying vec2 vUv;

    void main() {
      float zOverW = texture2D(tDepth, vUv).x;

      // Skip background pixels at maximum depth
      if (zOverW >= 1.0) {
        gl_FragColor = texture2D(tDiffuse, vUv);
        return;
      }

      // Convert current screen coordinates to Normalized Device Coordinates (NDC)
      vec4 currentNDC = vec4(vUv * 2.0 - 1.0, zOverW * 2.0 - 1.0, 1.0);

      // Reconstruct world position
      vec4 worldPos = uInverseViewProjectionMatrix * currentNDC;
      worldPos /= worldPos.w;

      // Project world position into previous frame's clip space
      vec4 prevClip = uPreviousViewProjectionMatrix * worldPos;
      vec4 prevNDC = prevClip / prevClip.w;

      // Compute velocity vector in UV space
      vec2 velocity = (currentNDC.xy - prevNDC.xy) * 0.5 * uVelocityScale;

      // Sample along the velocity vector
      vec4 color = texture2D(tDiffuse, vUv);
      vec2 coord = vUv;
      float totalWeight = 1.0;

      for (int i = 1; i < 16; ++i) {
        if (i >= uSamples) break;
        coord += velocity / float(uSamples);
        color += texture2D(tDiffuse, coord);
        totalWeight += 1.0;
      }

      gl_FragColor = color / totalWeight;
    }
  `
};

const motionBlurPass = new ShaderPass(MotionBlurShader);
motionBlurPass.uniforms.tDepth.value = renderTarget.depthTexture;
composer.addPass(motionBlurPass);

3. Track Projection Matrices

To supply the shader with accurate transform data, maintain instances of the current view-projection matrix, its inverse, and the previous frame's view-projection matrix:

const currentViewProjection = new THREE.Matrix4();
const previousViewProjection = new THREE.Matrix4();
const inverseViewProjection = new THREE.Matrix4();

let isFirstFrame = true;

function updateMatrices(camera) {
  // Compute current view-projection matrix (Projection * View)
  currentViewProjection.multiplyMatrices(camera.projectionMatrix, camera.matrixWorldInverse);
  inverseViewProjection.copy(currentViewProjection).invert();

  // If first run, initialize previous state to current
  if (isFirstFrame) {
    previousViewProjection.copy(currentViewProjection);
    isFirstFrame = false;
  }

  // Update uniforms
  motionBlurPass.uniforms.uInverseViewProjectionMatrix.value.copy(inverseViewProjection);
  motionBlurPass.uniforms.uPreviousViewProjectionMatrix.value.copy(previousViewProjection);
}

function postRenderMatrixUpdate() {
  // Store current matrix as previous for the upcoming frame
  previousViewProjection.copy(currentViewProjection);
}

4. Execute the Render Loop

Call the matrix calculations and composer render directly within the animation loop:

function animate() {
  requestAnimationFrame(animate);

  // Update camera controls or object animations
  controls.update();

  // Update camera world matrix and projection
  camera.updateMatrixWorld();

  // Set current matrices before rendering
  updateMatrices(camera);

  // Render post-processing pipeline
  composer.render();

  // Store matrix state for the next frame
  postRenderMatrixUpdate();
}

animate();

5. Adjusting Quality and Scale

  • uVelocityScale: Controls blur intensity. Increase this value for exaggerated stylistic blur or scale it according to delta time (clock.getDelta()) to ensure frame-rate-independent blur.
  • uSamples: Higher sample counts produce smoother trails but require more texture fetch operations. A count between 8 and 16 balances visual fidelity and performance.
