Three.js Volumetric Fog with Data3DTexture Raymarching

This article provides a complete guide on implementing volumetric fog and clouds in Three.js using Data3DTexture and custom raymarching shaders. You will learn how to generate procedural 3D noise, package it into a 3D texture, configure a bounding mesh container, and write GLSL raymarching logic to calculate light absorption and density accumulation.

1. Generating the 3D Noise Data

Raymarching volumetric clouds requires a continuous 3D density field. While 2D textures sample coordinates along \((u, v)\), a THREE.Data3DTexture samples across \((u, v, w)\), representing a 3D grid of voxel density values.

You can generate this dataset on the CPU using a typed array populated with 3D Perlin or Simplex noise:

import * as THREE from 'three';
import { createNoise3D } from 'simplex-noise';

function create3DNoiseTexture(size = 64) {
  const noise3D = createNoise3D();
  const data = new Uint8Array(size * size * size);
  let i = 0;

  for (let z = 0; z < size; z++) {
    for (let y = 0; y < size; y++) {
      for (let x = 0; x < size; x++) {
        // Normalize coordinates to [0, 1]
        const nx = x / size;
        const ny = y / size;
        const nz = z / size;

        // Sample noise with multiple octaves (Fractal Brownian Motion)
        let n = noise3D(nx * 4, ny * 4, nz * 4) * 0.5 + 0.5;
        n += (noise3D(nx * 8, ny * 8, nz * 8) * 0.5 + 0.5) * 0.5;
        n = Math.min(Math.max(n / 1.5, 0), 1);

        data[i++] = Math.floor(n * 255);
      }
    }
  }

  const texture = new THREE.Data3DTexture(data, size, size, size);
  texture.format = THREE.RedFormat;
  texture.type = THREE.UnsignedByteType;
  texture.minFilter = THREE.LinearFilter;
  texture.magFilter = THREE.LinearFilter;
  texture.wrapS = THREE.RepeatWrapping;
  texture.wrapT = THREE.RepeatWrapping;
  texture.wrapR = THREE.RepeatWrapping;
  texture.needsUpdate = true;

  return texture;
}

2. Setting Up the Bounding Container

A raymarching volume needs a geometric boundary within the Three.js scene graph. A standard unit THREE.BoxGeometry serves as the bounds:

const boxSize = 10;
const geometry = new THREE.BoxGeometry(boxSize, boxSize, boxSize);

// Compute bounding coordinates in world space for the ray intersection test
const boundsMin = new THREE.Vector3(-boxSize / 2, -boxSize / 2, -boxSize / 2);
const boundsMax = new THREE.Vector3(boxSize / 2, boxSize / 2, boxSize / 2);

const volumeMaterial = new THREE.ShaderMaterial({
  glslVersion: THREE.GLSL3,
  uniforms: {
    u_densityMap: { value: create3DNoiseTexture(64) },
    u_boundsMin: { value: boundsMin },
    u_boundsMax: { value: boundsMax },
    u_cameraPos: { value: new THREE.Vector3() },
    u_steps: { value: 64 },
    u_absorption: { value: 2.5 },
    u_densityScale: { value: 1.0 },
    u_lightColor: { value: new THREE.Color(1.0, 1.0, 1.0) }
  },
  vertexShader: vertexShaderSource,
  fragmentShader: fragmentShaderSource,
  side: THREE.BackSide, // Render backside so the camera can move inside the fog
  transparent: true,
  depthWrite: false
});

const volumeMesh = new THREE.Mesh(geometry, volumeMaterial);
scene.add(volumeMesh);

3. The Vertex Shader

The vertex shader calculates world-space vertex positions and passes them to the fragment stage to reconstruct ray directions.

out vec3 vWorldPosition;

void main() {
  vec4 worldPosition = modelMatrix * vec4(position, 1.0);
  vWorldPosition = worldPosition.xyz;
  gl_Position = projectionMatrix * viewMatrix * worldPosition;
}

4. The Fragment Shader (Raymarching Core)

The fragment shader computes the ray entry and exit intersections with the bounding box using the Slab method, marches along the ray segment, and integrates density using Beer's Law.

precision highp float;
precision highp sampler3D;

in vec3 vWorldPosition;
out vec4 fragColor;

uniform sampler3D u_densityMap;
uniform vec3 u_boundsMin;
uniform vec3 u_boundsMax;
uniform vec3 cameraPosition;
uniform int u_steps;
uniform float u_absorption;
uniform float u_densityScale;
uniform vec3 u_lightColor;

// Slab method to find entry and exit distances of the bounding box
vec2 hitBox(vec3 orig, vec3 dir) {
  vec3 invDir = 1.0 / dir;
  vec3 tMin = (u_boundsMin - orig) * invDir;
  vec3 tMax = (u_boundsMax - orig) * invDir;
  vec3 t1 = min(tMin, tMax);
  vec3 t2 = max(tMin, tMax);
  float tNear = max(max(t1.x, t1.y), t1.z);
  float tFar = min(min(t2.x, t2.y), t2.z);
  return vec2(tNear, tFar);
}

void main() {
  vec3 rayOrigin = cameraPosition;
  vec3 rayDir = normalize(vWorldPosition - rayOrigin);

  vec2 bounds = hitBox(rayOrigin, rayDir);
  if (bounds.x > bounds.y || bounds.y < 0.0) {
    discard;
  }

  bounds.x = max(bounds.x, 0.0); // Clamp to zero if the camera is inside the box

  float distance = bounds.y - bounds.x;
  float stepLength = distance / float(u_steps);
  vec3 stepVec = rayDir * stepLength;
  vec3 currentPos = rayOrigin + rayDir * bounds.x;

  float transmittance = 1.0;
  vec3 accumulatedColor = vec3(0.0);

  for (int i = 0; i < u_steps; i++) {
    // Map current world position to local UVW [0, 1] coordinates
    vec3 uvw = (currentPos - u_boundsMin) / (u_boundsMax - u_boundsMin);

    // Sample the 3D texture
    float density = texture(u_densityMap, uvw).r * u_densityScale;

    if (density > 0.0) {
      // Calculate light attenuation using Beer-Lambert law
      float opticalDepth = density * stepLength * u_absorption;
      float stepTransmittance = exp(-opticalDepth);

      // Accumulate color
      accumulatedColor += u_lightColor * density * (1.0 - stepTransmittance) * transmittance;
      transmittance *= stepTransmittance;

      // Early exit if the ray is fully saturated
      if (transmittance < 0.01) {
        break;
      }
    }

    currentPos += stepVec;
  }

  fragColor = vec4(accumulatedColor, 1.0 - transmittance);
}

5. Performance Optimizations

  1. Jittering (Dithering): Uniform ray steps often cause wood-grain or slicing artifacts. Introduce a randomized offset to the ray starting distance using a screen-space blue noise texture or interleaved pseudo-random function: 
    float jitter = fract(sin(dot(gl_FragCoord.xy, vec2(12.9898, 78.233))) * 43758.5453);
currentPos = rayOrigin + rayDir * (bounds.x + jitter * stepLength);
  2. Dynamic Step Resolution: Increase u_steps when viewing up close, and decrease it dynamically when the bounding mesh is distant from the camera.
  3. Half-Resolution Rendering: Render the raymarched volume to an offscreen render target at half or quarter screen resolution, then upscale and composite it onto the primary scene buffer with a bilateral depth-aware filter.
