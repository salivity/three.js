Reset SkinnedMesh to Bind Pose in Three.js

Resetting a SkinnedMesh to its original bind pose is a common requirement in Three.js when interrupting animations, transitioning characters to a neutral state, or switching between procedural and keyframed motion. This guide demonstrates how to programmatically restore a skeleton to its initial rest state, covering both an immediate snap using the built-in skeleton API and a smooth interpolation method to eliminate visual popping.

Immediate Reset: The Built-In Method

The quickest way to revert a SkinnedMesh to its rest pose is using the pose() method on its Skeleton. When an AnimationMixer is active, it updates the bones every frame, so any active animation actions must be stopped or disabled first.

function snapToBindPose(skinnedMesh, mixer) {
    // 1. Stop active animations so they do not override the pose
    if (mixer) {
        mixer.stopAllAction();
    }

    // 2. Reset bones to their bind pose positions, rotations, and scales
    skinnedMesh.skeleton.pose();

    // 3. Update the mesh's bone matrices
    skinnedMesh.skeleton.update();
}

Calling skeleton.pose() iterates through every bone in the skeleton and sets its local position, quaternion, and scale to match the transformation values defined by the skeleton's boneInverses array.

Smooth Transition: Interpolating to Bind Pose

An instant snap can create an unnatural, jarring visual jump. To transition smoothly over time, you can capture the target bind transforms and interpolate (lerp and slerp) the current bone transforms back to them within your render loop.

1. Cache the Bind Transforms

Retrieve the bind pose transforms once and store them for reference:

function getBindTransforms(skinnedMesh) {
    const skeleton = skinnedMesh.skeleton;
    
    // Temporarily apply bind pose to read the rest values
    skeleton.pose();

    const bindTransforms = skeleton.bones.map((bone) => ({
        position: bone.position.clone(),
        quaternion: bone.quaternion.clone(),
        scale: bone.scale.clone()
    }));

    return bindTransforms;
}

2. Transition Over Time

To animate the transition, deactivate the AnimationMixer so it stops updating bones, then interpolate each bone's current transform toward the stored bind transform using Vector3.lerp and Quaternion.slerp.

class BindPoseTransition {
    constructor(skinnedMesh, duration = 0.5) {
        this.skinnedMesh = skinnedMesh;
        this.skeleton = skinnedMesh.skeleton;
        this.duration = duration;
        this.elapsed = 0;
        this.isTransitioning = false;

        // Cache target bind transforms
        this.targetTransforms = getBindTransforms(skinnedMesh);
        this.startTransforms = [];
    }

    start(mixer) {
        if (mixer) {
            mixer.stopAllAction();
        }

        // Snapshot current bone transforms at the start of transition
        this.startTransforms = this.skeleton.bones.map((bone) => ({
            position: bone.position.clone(),
            quaternion: bone.quaternion.clone(),
            scale: bone.scale.clone()
        }));

        this.elapsed = 0;
        this.isTransitioning = true;
    }

    update(delta) {
        if (!this.isTransitioning) return;

        this.elapsed += delta;
        const progress = Math.min(this.elapsed / this.duration, 1.0);

        // Interpolate each bone
        for (let i = 0; i < this.skeleton.bones.length; i++) {
            const bone = this.skeleton.bones[i];
            const start = this.startTransforms[i];
            const target = this.targetTransforms[i];

            bone.position.lerpVectors(start.position, target.position, progress);
            bone.quaternion.slerpQuaternions(start.quaternion, target.quaternion, progress);
            bone.scale.lerpVectors(start.scale, target.scale, progress);
        }

        this.skeleton.update();

        if (progress >= 1.0) {
            this.isTransitioning = false;
            // Ensure exact precision at completion
            this.skeleton.pose();
            this.skeleton.update();
        }
    }
}

3. Execution in the Animation Loop

Integrate the transition instance inside your standard render/tick loop:

const transitionHandler = new BindPoseTransition(skinnedMesh, 0.4);

// Trigger the transition when required
transitionHandler.start(mixer);

function animate() {
    requestAnimationFrame(animate);
    
    const delta = clock.getDelta();

    if (transitionHandler.isTransitioning) {
        transitionHandler.update(delta);
    } else {
        mixer.update(delta);
    }

    renderer.render(scene, camera);
}

Using this approach, existing animations gracefully blend toward the bind pose without creating sudden visual artifacts.
