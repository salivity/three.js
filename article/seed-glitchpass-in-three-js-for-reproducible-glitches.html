Seed GlitchPass in Three.js for Reproducible Glitches

This article demonstrates how to achieve deterministic, reproducible digital glitch patterns in Three.js by replacing the default pseudo-random behavior in GlitchPass. By default, GlitchPass relies on native Math.random() to generate its displacement heightmaps and schedule glitch triggers. By overriding these internal methods with a seeded pseudo-random number generator (PRNG), you can control when, where, and how visual artifacts appear, ensuring identical post-processing results across any session or render pass.

Understanding GlitchPass Randomness

The standard GlitchPass (three/addons/postprocessing/GlitchPass.js) introduces randomness in two primary areas:

  1. Displacement Texture (generateHeightmap): Generates a dynamic DataTexture filled with randomized values to displace UV coordinates in the shader.
  2. Timing and Burst Logic (generateTrigger): Evaluates curF (current frame duration) and random step intervals to toggle between subtle glitches and heavy, disruptive visual spikes.

Because both routines call Math.random(), the artifacts and their intervals are completely non-deterministic by default.

Step 1: Implement a Seeded PRNG

To make random output deterministic, create a small PRNG algorithm such as Mulberry32, which generates 32-bit pseudo-random floating-point numbers between 0 and 1 based on an integer seed.

function createPRNG(seed) {
    let s = seed >>> 0;
    return function() {
        s = (s + 0x6D2B79F5) | 0;
        let t = Math.imul(s ^ (s >>> 15), 1 | s);
        t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
        return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
}

Step 2: Extend GlitchPass with Custom Seed Logic

Subclass GlitchPass to accept an initial seed and replace Math.random with your seeded PRNG in generateTrigger and generateHeightmap.

import * as THREE from 'three';
import { GlitchPass } from 'three/addons/postprocessing/GlitchPass.js';

class SeededGlitchPass extends GlitchPass {
    constructor(dt_size = 64, seed = 12345) {
        super(dt_size);
        this.seed = seed;
        this.rng = createPRNG(this.seed);
        
        // Re-generate initial heightmap with the seeded PRNG
        this.uniforms['tDisp'].value = this.generateHeightmap(dt_size);
    }

    setSeed(seed) {
        this.seed = seed;
        this.rng = createPRNG(seed);
        this.curF = 0;
        this.generateTrigger();
        this.uniforms['tDisp'].value = this.generateHeightmap(64);
    }

    generateTrigger() {
        this.randX = THREE.MathUtils.randInt(120, 240); // Uses internal random by default; override with PRNG:
        this.randX = Math.floor(this.rng() * (240 - 120 + 1)) + 120;
    }

    generateHeightmap(dt_size) {
        const data_arr = new Float32Array(dt_size * dt_size);
        const length = dt_size * dt_size;

        for (let i = 0; i < length; i++) {
            const val = this.rng(); // Seeded noise value
            data_arr[i] = val;
        }

        const texture = new THREE.DataTexture(
            data_arr,
            dt_size,
            dt_size,
            THREE.RedFormat,
            THREE.FloatType
        );
        texture.needsUpdate = true;
        return texture;
    }
}

Step 3: Override the Render Loop Execution

During each frame in the render function of GlitchPass, standard Three.js implementations make additional non-deterministic calculations (such as Math.random() < 0.05 for periodic perturbation). Override the render method to ensure all intermediate randomization uses this.rng:

SeededGlitchPass.prototype.render = function (renderer, writeBuffer, readBuffer, deltaTime, maskActive) {
    const factor = Math.max(readBuffer.width, readBuffer.height);
    this.uniforms['tDiffuse'].value = readBuffer.texture;
    this.uniforms['seed'].value = this.rng(); // Pass deterministic random seed to shader uniform
    this.uniforms['byp'].value = 0;

    if (this.curF % this.randX === 0 || this.goWild === true) {
        this.uniforms['amount'].value = this.rng() / 30;
        this.uniforms['angle'].value = THREE.MathUtils.degToRad(this.rng() * 360);
        this.uniforms['distortion_x'].value = (this.rng() - 0.5) / 5;
        this.uniforms['distortion_y'].value = (this.rng() - 0.5) / 5;
        this.uniforms['seed_x'].value = (this.rng() - 0.5) / factor;
        this.uniforms['seed_y'].value = (this.rng() - 0.5) / factor;
        this.curF = 0;
        this.generateTrigger();
    } else if (this.curF % this.randX < this.randX / 5) {
        this.uniforms['amount'].value = this.rng() / 90;
        this.uniforms['angle'].value = THREE.MathUtils.degToRad(this.rng() * 360);
        this.uniforms['distortion_x'].value = (this.rng() - 0.5) / 20;
        this.uniforms['distortion_y'].value = (this.rng() - 0.5) / 20;
        this.uniforms['seed_x'].value = (this.rng() - 0.5) / factor;
        this.uniforms['seed_y'].value = (this.rng() - 0.5) / factor;
    } else if (!this.goWild) {
        this.uniforms['byp'].value = 1;
    }

    this.curF++;

    if (this.renderToScreen) {
        renderer.setRenderTarget(null);
        this.fsQuad.render(renderer);
    } else {
        renderer.setRenderTarget(writeBuffer);
        if (this.clear) renderer.clear();
        this.fsQuad.render(renderer);
    }
};

Reproducing Patterns

To reset the effect back to its exact initial pattern—for instance, when rewinding a timeline, exporting a canvas animation frame-by-frame, or synchronizing multiple clients:

const glitchPass = new SeededGlitchPass(64, 42);
composer.addPass(glitchPass);

// Reset back to original deterministic sequence at frame 0
function resetGlitchTimeline(targetSeed = 42) {
    glitchPass.setSeed(targetSeed);
}

By substituting Three.js's native Math.random invocations with a managed generator instance inside GlitchPass, every displacement calculation, glitch frame interval, and perturbation vector follows a strictly reproducible sequence.
