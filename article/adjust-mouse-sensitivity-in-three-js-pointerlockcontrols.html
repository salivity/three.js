Adjust Mouse Sensitivity in Three.js PointerLockControls

This guide explains how to independently configure horizontal and vertical mouse sensitivity when using PointerLockControls in Three.js. While the built-in PointerLockControls implementation provides a global pointerSpeed property to scale general responsiveness, it does not natively provide independent axes controls. By customizing the mouse movement event handler or adjusting the camera's rotational Euler angles manually, you can achieve granular control over both horizontal (yaw) and vertical (pitch) sensitivity.

The Limitation of Default Controls

By default, Three.js applies a hardcoded rotation delta multiplied by pointerSpeed:

controls.pointerSpeed = 1.0; // Scales both X and Y uniformly

Because event.movementX and event.movementY are read-only properties of the native browser MouseEvent, you cannot modify them directly before Three.js receives them. To implement separate X and Y sensitivities, you must either override the default onMouseMove callback or write a custom pointer lock listener.

Method 1: Overriding the onMouseMove Handler

The most direct way to achieve separate sensitivities without rewriting the entire control system is to disconnect the default event listener and bind a customized mouse movement handler to the document.

import * as THREE from 'three';
import { PointerLockControls } from 'three/addons/controls/PointerLockControls.js';

// Initialize camera and controls
const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
const controls = new PointerLockControls(camera, document.body);

// Define independent sensitivity factors
const sensitivity = {
    horizontal: 0.002, // Mouse X / Camera Yaw
    vertical: 0.0015   // Mouse Y / Camera Pitch
};

// Internal Euler angles for clamping pitch
const euler = new THREE.Euler(0, 0, 0, 'YXZ');
const minPolarAngle = 0; // Look straight up limit
const maxPolarAngle = Math.PI; // Look straight down limit

function onMouseMoveCustom(event) {
    if (!controls.isLocked) return;

    const movementX = event.movementX || event.mozMovementX || event.webkitMovementX || 0;
    const movementY = event.movementY || event.mozMovementY || event.webkitMovementY || 0;

    const camera = controls.camera;
    euler.setFromQuaternion(camera.quaternion);

    // Apply horizontal sensitivity to the yaw (Y axis)
    euler.y -= movementX * sensitivity.horizontal;

    // Apply vertical sensitivity to the pitch (X axis)
    euler.x -= movementY * sensitivity.vertical;

    // Clamp vertical view angle to prevent camera flipping
    euler.x = Math.max(Math.PI / 2 - maxPolarAngle, Math.min(Math.PI / 2 - minPolarAngle, euler.x));

    camera.quaternion.setFromEuler(euler);
    controls.dispatchEvent({ type: 'change' });
}

// Disconnect the built-in mouse listener and bind the custom one
controls.disconnect();
document.addEventListener('mousemove', onMouseMoveCustom, false);

Method 2: Subclassing PointerLockControls

For modular or larger codebases, subclassing provides a clean, reusable component that exposes sensitivity settings via standard class properties.

import * as THREE from 'three';
import { PointerLockControls } from 'three/addons/controls/PointerLockControls.js';

class CustomPointerLockControls extends PointerLockControls {
    constructor(camera, domElement) {
        super(camera, domElement);

        this.sensitivityX = 0.002;
        this.sensitivityY = 0.002;
        this.minPolarAngle = 0;
        this.maxPolarAngle = Math.PI;

        const _euler = new THREE.Euler(0, 0, 0, 'YXZ');

        // Replace the internal onMouseMove handler
        this._onMouseMove = (event) => {
            if (!this.isLocked) return;

            const movementX = event.movementX || 0;
            const movementY = event.movementY || 0;

            _euler.setFromQuaternion(this.camera.quaternion);

            _euler.y -= movementX * this.sensitivityX;
            _euler.x -= movementY * this.sensitivityY;

            _euler.x = Math.max(
                Math.PI / 2 - this.maxPolarAngle, 
                Math.min(Math.PI / 2 - this.minPolarAngle, _euler.x)
            );

            this.camera.quaternion.setFromEuler(_euler);
            this.dispatchEvent({ type: 'change' });
        };
    }

    connect() {
        super.connect();
        // Remove the base class handler and attach the custom handler
        this.domElement.ownerDocument.removeEventListener('mousemove', this.onMouseMove);
        this.domElement.ownerDocument.addEventListener('mousemove', this._onMouseMove);
    }

    disconnect() {
        super.disconnect();
        this.domElement.ownerDocument.removeEventListener('mousemove', this._onMouseMove);
    }
}

Implementation Usage

Once the custom control class or override is defined, instantiate it and adjust sensitivity dynamically based on user settings or preferences:

const playerControls = new CustomPointerLockControls(camera, document.body);

// Adjust independently via UI sliders or settings
playerControls.sensitivityX = 0.0025; // Faster turning
playerControls.sensitivityY = 0.0012; // Slower vertical aiming

// Invert vertical aim if required
const invertY = false;
if (invertY) {
    playerControls.sensitivityY *= -1;
}

// Lock pointer on user interaction
document.addEventListener('click', () => {
    playerControls.lock();
});

Using this setup decouples yaw and pitch speeds completely, avoids native event conflicts, and ensures proper polar angle clamping so the camera does not flip when looking straight up or down.
